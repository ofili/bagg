The existing integration contains five email methods: invoice delivery, quote delivery, a question on a quote, invoice reminders, and sign-in codes. The implementation below keeps those method signatures intact, so existing callers should not need to change.

The important change is that all five email types use the same shared Bagg layout, with their own message content and action. The SMTP path will send both HTML and plain-text alternatives. The Sendly path will receive both formats as well.

def _escape(value: object) -> str:
    """Escape dynamic values before inserting them into HTML."""
    return escape(str(value), quote=True)


def _safe_url(value: str) -> str:
    """Allow only absolute HTTP(S) links in email actions."""
    try:
        parsed = urlsplit(value)
    except (TypeError, ValueError):
        return "#"

    if parsed.scheme not in {"https", "http"} or not parsed.netloc:
        return "#"

    return escape(value, quote=True)


def _safe_subject(value: str) -> str:
    """Prevent user-controlled values from injecting email headers."""
    return re.sub(r"[\r\n]+", " ", str(value)).strip()

def _plain_text(
    *,
    title: str,
    content: str,
    cta_label: Optional[str] = None,
    cta_url: Optional[str] = None,
    note: Optional[str] = None,
) -> str:
    """Produce a readable plain-text alternative for the HTML template."""
    parts = [
        "BAGG",
        "Invoice like a pro. Get paid without chasing.",
        "",
        title,
        "",
        content.strip(),
    ]

    if cta_label and cta_url:
        parts.extend(["", f"{cta_label}: {cta_url}"])

    if note:
        parts.extend(["", note.strip()])

    parts.extend(["", "Bagg · Business, handled."])
    return "\n".join(parts)


class EmailSender:
    def _sendly_from(self) -> str:
        address = settings.SENDLY_FROM_EMAIL or settings.DEFAULT_FROM_EMAIL
        if settings.SENDLY_FROM_NAME:
            return f"{settings.SENDLY_FROM_NAME} <{address}>"
        return address

    async def _send_via_sendly(
        self,
        to_email: str,
        subject: str,
        html: str,
        text: str,
    ) -> bool:
        try:
            async with httpx.AsyncClient(timeout=15.0) as client:
                response = await client.post(
                    SENDLY_API_URL,
                    headers={
                        "Authorization": f"Bearer {settings.SENDLY_API_KEY}",
                        "Content-Type": "application/json",
                    },
                    json={
                        "channel": "email",
                        "to": to_email,
                        "from": self._sendly_from(),
                        "subject": _safe_subject(subject),
                        "html": html,
                        "text": text,
                    },
                )

            if 200 <= response.status_code < 300:
                logger.info("Email accepted by SendlyAI for %s", to_email)
                return True

            logger.warning(
                "SendlyAI rejected email to %s: %s",
                to_email,
                response.status_code,
            )
            return False
        except Exception:
       	    # Avoid logging message bodies, codes, or provider credentials.
            logger.exception("SendlyAI email delivery failed for %s", to_email)
            return False

    def _send_via_smtp(
        self,
        to_email: str,
        subject: str,
        html: str,
        text: str,
    ) -> bool:
        msg = MIMEMultipart("alternative")
        msg["Subject"] = _safe_subject(subject)
        msg["From"] = self._sendly_from()
        msg["To"] = to_email

        # Plain text first, HTML second: standard multipart/alternative order.
        msg.attach(MIMEText(text, "plain", "utf-8"))
        msg.attach(MIMEText(html, "html", "utf-8"))

        try:
            with smtplib.SMTP(settings.SMTP_HOST, settings.SMTP_PORT) as server:
                if settings.SMTP_USER and settings.SMTP_PASSWORD:
                    server.login(settings.SMTP_USER, settings.SMTP_PASSWORD)
                server.sendmail(msg["From"], [to_email], msg.as_string())

            logger.info("Email accepted by SMTP for %s", to_email)
            return True
        except Exception:
            if settings.ENVIRONMENT.strip().lower() == "production":
                logger.exception("SMTP email delivery failed for %s", to_email)
                return False

            logger.info("SMTP unavailable in development; email not delivered")
            return True

    async def _deliver(
        self,
        to_email: str,
        subject: str,
        html: str,
        text: str,
    ) -> bool:
        if settings.SENDLY_API_KEY:
            return await self._send_via_sendly(to_email, subject, html, text)
        return self._send_via_smtp(to_email, subject, html, text)

## 4. Two integration details to verify before deploying

1. Sendly payload compatibility. The existing integration sends `html`; the replacement also sends `text`. Verify that Sendly accepts the additional field. SMTP already uses standard `multipart/alternative` with both formats.
2. Development delivery semantics. The existing implementation treats unavailable local SMTP as successful in non-production. The replacement preserves that return behaviour, but logs that the email was not delivered. Do not interpret a `True` result in that development fallback as proof that an email reached a mailbox.
